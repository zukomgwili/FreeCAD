cmake_minimum_required(VERSION 3.20)
project(freecad_cocoa_accessibility_regression LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "This regression requires the native macOS Cocoa platform plugin")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11.2 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(selector-repro main.mm)
target_compile_definitions(selector-repro PRIVATE QT_NO_KEYWORDS)
target_link_libraries(selector-repro PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets "-framework AppKit")
target_compile_options(selector-repro PRIVATE -Wall -Wextra)
